// ------------------------------------------------------------------ Fungsi waktu (ET)
f_mod(int t) => hour(t, TZ) * 60 + minute(t, TZ)
f_off(int t) => ((f_mod(t) - wStart) % 1440 + 1440) % 1440
f_inWin(int t) => f_off(t) < wLen
f_qual(int barOpen) =>
    int o = f_off(barOpen)
    o < wLen and o + barMin <= sigOff
f_winId(int t) =>
    int w0 = timestamp(TZ, year(t, TZ), month(t, TZ), dayofmonth(t, TZ), wsH, wsM)
    t < w0 ? w0 - 86400000 : w0
// hari dagang server (NY close): mulai 17:00 ET
f_dayStart(int t) =>
    int d0 = timestamp(TZ, year(t, TZ), month(t, TZ), dayofmonth(t, TZ), 17, 0)
    t < d0 ? d0 - 86400000 : d0
f_atET(int t, int h, int m) => timestamp(TZ, year(t, TZ), month(t, TZ), dayofmonth(t, TZ), h, m)
f_hm(int t, string tz) => str.format("{0,number,00}:{1,number,00}", hour(t, tz), minute(t, tz))
f_dt(int t) => str.format("{0,number,0000}-{1,number,00}-{2,number,00} {3}", year(t, TZ), month(t, TZ), dayofmonth(t, TZ), f_hm(t, TZ))
f_px(float v) => na(v) ? "n/a" : str.tostring(v, format.mintick)

// bulatkan ke angka "bagus" 1-2-3-5-10 × 10^n
f_nice(float x) =>
    float e = math.pow(10, math.floor(math.log10(x)))
    float m = x / e
    float n = m < 1.5 ? 1.0 : m < 2.5 ? 2.0 : m < 4.0 ? 3.0 : m < 7.5 ? 5.0 : 10.0
    n * e

f_addEvents(string src) =>
    int added = 0
    for ln in str.split(src, "\n")
        string s = str.trim(ln)
        array<string> parts = str.split(s, " ")
        if parts.size() >= 3
            array<string> d  = str.split(parts.get(0), "-")
            array<string> hm = str.split(parts.get(1), ":")
            if d.size() == 3 and hm.size() == 2
                int ts = timestamp(TZ, int(str.tonumber(d.get(0))), int(str.tonumber(d.get(1))), int(str.tonumber(d.get(2))),
                     int(str.tonumber(hm.get(0))), int(str.tonumber(hm.get(1))))
                string k = str.upper(parts.get(2))
                evT.push(ts)
                evK.push(str.contains(k, "FOMC") ? 2 : 1)
                evN.push(k)
                added += 1
    added

// event CPI/NFP terakhir dan status FOMC di hari dagang server yang memuat t
f_yudha(int t) =>
    int ds = f_dayStart(t)
    int de = ds + 86400000
    int lastCN = 0
    bool fomc = false
    string names = ""
    for i = 0 to evT.size() - 1
        int et = evT.get(i)
        if et >= ds and et < de
            if evK.get(i) == 1
                lastCN := math.max(lastCN, et)
            else
                fomc := true
            names += (names == "" ? "" : ", ") + evN.get(i) + " " + f_hm(et, TZ)
    [lastCN, fomc, names]

f_break(int pp, int cc) =>
    int brk = 0
    int lvl = 0
    if cc > pp
        int t = gridPts * math.floor(float(cc - 1) / gridPts)
        if t >= pp
            brk := 1
            lvl := t
    else if cc < pp
        int t = gridPts * math.ceil(float(cc + 1) / gridPts)
        if t <= pp
            brk := -1
            lvl := t
    [brk, lvl]
